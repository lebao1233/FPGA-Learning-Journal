
module uart_rx_tb();
  reg tb_clk;
  reg tb_i_rx_serial;
  reg rst_n;
  wire tb_o_rx_dv;
  wire [7:0] tb_o_rx_serial;
  uart_rx DUT(
    .i_rx_serial(tb_i_rx_serial),
    .rst_n(rst_n),
    .clk(tb_clk),
    .o_rx_dv(tb_o_rx_dv),
    .o_rx_serial(tb_o_rx_serial));
  always begin 
    #5 tb_clk=~tb_clk; end
     initial begin
    $dumpfile("dump.vcd");
       $dumpvars(0, uart_rx_tb);
       tb_i_rx_serial=1;
       tb_clk=0;
       rst_n=0;
      #50;
        rst_n = 1;
     #30;  
  tb_i_rx_serial=0;//start
       $display("startbit at %0t", $time);
       repeat(234)@(posedge tb_clk);
     
    tb_i_rx_serial=0;
       $display("bit0 at %0t", $time);
       repeat(234)@(posedge tb_clk);
  
    tb_i_rx_serial=1;
       $display("bit1 at %0t", $time);
       repeat(234)@(posedge tb_clk);
    
    tb_i_rx_serial=0;
       $display("bit2 at %0t", $time);
       repeat(234)@(posedge tb_clk);
  
    tb_i_rx_serial=0;
       $display("bit3 at %0t", $time);
       repeat(234)@(posedge tb_clk);
  
     tb_i_rx_serial=0;
       $display("bit4 at %0t", $time);
       repeat(234)@(posedge tb_clk);
  
     tb_i_rx_serial=1;
       $display("bit5 at %0t", $time);
       repeat(234)@(posedge tb_clk);
  
     tb_i_rx_serial=1;
       $display("bit6 at %0t", $time);
       repeat(234)@(posedge tb_clk);
  
     tb_i_rx_serial=0;
       $display("bit7 at %0t", $time);
       repeat(234)@(posedge tb_clk);
    
    tb_i_rx_serial=1;
        repeat(234)@(posedge tb_clk);
       repeat(20)@(posedge tb_clk);
    
       
  $finish;
     end
endmodule
